\begin{guidance}
\textbf{General Submission Guidelines:}
\begin{itemize}
    \item \textbf{Page Limit:} The technical report must not exceed \textbf{10 pages}, including all figures, tables, and references.
    \item \textbf{Submission Deadline:} 2026-08-06 14:00 (KST, UTC+09:00)
    \item \textbf{Submission Method:} Submit the report via email to \texttt{techreport@optichallenge.com}. Please include your team name in the email subject and file name, following the format below:
\begin{itemize}
\item Email subject: [OGC2026] TeamName
\item Technical report: [OGC2026\_TechReport]\_TeamName.pdf
\end{itemize}
    \item \textbf{Format Requirement:} The final submission must be in PDF format. Do not modify the formatting settings of the provided template.
    \item \textbf{Final PDF Preparation:} All Writing Guidelines boxes must be completely removed before submission.
    \item \textbf{Blind Review:} The reports will be evaluated anonymously. Do not include any information in the report that could identify the team, including team name, affiliations,  repository links, or other identifying details.
\end{itemize}
\end{guidance}
